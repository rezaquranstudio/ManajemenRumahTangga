                    return d.toLocaleDateString('id-ID', { weekday: 'long', day: 'numeric', month: 'long', year: 'numeric' });
                },

                get greetingText() {
                    const hrs = new Date().getHours();
                    if (hrs < 11) return 'Selamat Pagi';
                    if (hrs < 15) return 'Selamat Siang';
                    if (hrs < 18) return 'Selamat Sore';
                    return 'Selamat Malam';
                },

                formatIDR(val) {
                    return new Intl.NumberFormat('id-ID', { style: 'currency', currency: 'IDR', maximumFractionDigits: 0 }).format(val || 0);
                },

                formatDate(str) {
                    if (!str) return '-';
                    const parts = str.split('-');
                    if (parts.length !== 3) return str;
                    return `${parts[2]}/${parts[1]}/${parts[0]}`;
                },

                calculateTotalAssetValue() {
                    return this.inventory.reduce((sum, item) => sum + (Number(item.purchasePrice) || 0), 0);
                },

                calculateUnpaidBillsTotal() {
                    return this.bills.filter(b => b.status !== 'Sudah Dibayar').reduce((sum, b) => sum + (Number(b.amount) || 0), 0);
                },

                calculateTotalVehicleServiceCost() {
                    return this.services.reduce((sum, s) => sum + (Number(s.cost) || 0), 0);
                },

                calculateMonthlyExpenses() {
                    return this.transactions.filter(t => t.type === 'Pengeluaran').reduce((sum, t) => sum + (Number(t.amount) || 0), 0);
                },

                calculateMonthlyIncome() {
                    return this.transactions.filter(t => t.type === 'Pemasukan').reduce((sum, t) => sum + (Number(t.amount) || 0), 0);
                },

                getBudgetUsagePercent() {
                    const exp = this.calculateMonthlyExpenses();
                    return Math.min(Math.round((exp / this.monthlyBudgetLimit) * 100), 100);
                },

                getPendingVehicleTaxCount() {
                    return this.vehicles.length;
                },

                // Filters
                get filteredInventory() {
                    return this.inventory.filter(item => {
                        const matchesSearch = item.name.toLowerCase().includes(this.inventorySearch.toLowerCase()) ||
                                              (item.brand && item.brand.toLowerCase().includes(this.inventorySearch.toLowerCase())) ||
                                              (item.location && item.location.toLowerCase().includes(this.inventorySearch.toLowerCase()));
                        const matchesCategory = !this.inventoryCategoryFilter || item.category === this.inventoryCategoryFilter;
                        const matchesAsset = !this.inventoryAssetFilter || (this.inventoryAssetFilter === 'asset' ? item.isAsset : !item.isAsset);
                        return matchesSearch && matchesCategory && matchesAsset;
                    });
                },

                getSmartReminders() {
                    const list = [];
                    // Unpaid bills
                    this.bills.filter(b => b.status !== 'Sudah Dibayar').forEach(b => {
                        list.push({
                            title: `Tagihan: ${b.title}`,
                            subtitle: `Nominal ${this.formatIDR(b.amount)}`,
                            dueDateText: `Tgl ${b.dueDateDay}`,
                            urgency: b.dueDateDay <= 15 ? 'high' : 'medium'
                        });
                    });
                    // Vehicle taxes
                    this.vehicles.forEach(v => {
                        list.push({
                            title: `Pajak: ${v.name}`,
                            subtitle: `Plat ${v.licensePlate}`,
                            dueDateText: this.formatDate(v.taxDueDate),
                            urgency: 'medium'
                        });
                    });
                    // Maintenance
                    this.maintenanceTasks.filter(m => m.status === 'Terjadwal').forEach(m => {
                        list.push({
                            title: `Perawatan: ${m.title}`,
                            subtitle: m.location,
                            dueDateText: this.formatDate(m.nextDueDate),
                            urgency: 'low'
                        });
                    });
                    return list;
                },

                getServicesForVehicle(vId) {
                    return this.services.filter(s => s.vehicleId === vId);
                },

                getSpentForCategory(cat) {
                    return this.transactions.filter(t => t.type === 'Pengeluaran' && t.category === cat).reduce((s, t) => s + (Number(t.amount) || 0), 0);
                },

                getCategoryBudgetPercent(b) {
                    const spent = this.getSpentForCategory(b.category);
                    return Math.round((spent / b.limit) * 100);
                },

                getAllCalendarEvents() {
                    const events = [];
                    this.bills.forEach(b => {
                        events.push({ title: b.title, category: 'Tagihan Rutin', subtitle: this.formatIDR(b.amount), monthText: 'NOV', dayText: b.dueDateDay, type: 'Bill' });
                    });
                    this.vehicles.forEach(v => {
                        events.push({ title: `Pajak ${v.name}`, category: 'Pajak Kendaraan', subtitle: v.licensePlate, monthText: 'NOV', dayText: '25', type: 'Tax' });
                    });
                    this.maintenanceTasks.forEach(m => {
                        events.push({ title: m.title, category: 'Perawatan Rumah', subtitle: m.location, monthText: 'DES', dayText: '01', type: 'Maintenance' });
                    });
                    return events;
                },

                getSearchResults() {
                    if (!this.globalSearchQuery.trim()) return [];
                    const q = this.globalSearchQuery.toLowerCase();
                    const results = [];

                    this.inventory.forEach(i => {
                        if (i.name.toLowerCase().includes(q)) {
                            results.push({ id: i.id, title: i.name, subtitle: `Inventaris • ${i.location}`, type: 'Barang', tab: 'inventory' });
                        }
                    });
                    this.vehicles.forEach(v => {
                        if (v.name.toLowerCase().includes(q) || v.licensePlate.toLowerCase().includes(q)) {
                            results.push({ id: v.id, title: v.name, subtitle: `Kendaraan • ${v.licensePlate}`, type: 'Motor/Mobil', tab: 'vehicles' });
                        }
                    });
                    this.documents.forEach(d => {
                        if (d.title.toLowerCase().includes(q)) {
                            results.push({ id: d.id, title: d.title, subtitle: `Dokumen • ${d.category}`, type: 'Dokumen', tab: 'documents' });
                        }
                    });

                    return results;
                },

                saveInventoryItem() {
                    if (this.inventoryForm.id) {
                        const idx = this.inventory.findIndex(i => i.id === this.inventoryForm.id);
                        if (idx !== -1) this.inventory[idx] = { ...this.inventoryForm };
                    } else {
                        this.inventory.push({ ...this.inventoryForm, id: 'inv-' + Date.now() });
                    }
                    this.saveAllData();
                    this.showInventoryModal = false;
                },

                    this.activeQRItem = item;
                    this.showQRModal = true;
                },

                confirmDelete(type, id) {
                    this.deleteTarget = { type, id };
                    this.showDeleteConfirmModal = true;
                },

                executeDelete() {
                    const { type, id } = this.deleteTarget;
                    if (type === 'inventory') this.inventory = this.inventory.filter(i => i.id !== id);
                    if (type === 'maintenance') this.maintenanceTasks = this.maintenanceTasks.filter(m => m.id !== id);
                    if (type === 'bill') this.bills = this.bills.filter(b => b.id !== id);
                    if (type === 'document') this.documents = this.documents.filter(d => d.id !== id);
                    
                    this.saveAllData();
                    this.showDeleteConfirmModal = false;
                },

                },

                handleRestoreFile(event) {
                    const file = event.target.files[0];
                    if (!file) return;
                    const reader = new FileReader();
                    reader.onload = (e) => {
                        try {
                            const data = JSON.parse(e.target.result);
                            Object.keys(data).forEach(k => localStorage.setItem(k, data[k]));
                            this.loadData();
                            alert('Data HOMEBASE Berhasil Dipulihkan!');
                        } catch(err) {
                            alert('Format File Backup Tidak Valid!');
                        }
                    };
                    reader.readAsText(file);
                },
